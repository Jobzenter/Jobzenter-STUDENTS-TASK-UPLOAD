Example:

Input: "hello"
Output: "olleh"
--------------------------------------

--------
public class ReverseString{
    public static void main(String[] args){
        String value = "Hello";
        String rev = " ";
        for(int i = value.length()-1; i>=0; i--){
            rev += value.charAt(i);
        }
        System.out.println("Reversed String:"+rev);
    }
}

3.Description: Write a function to check if a given word or number is a palindrome (reads the same forward and backward).
Example:

Input: "racecar"
Output: true

Input: "hello"
Output: false
---------------------------------------------------------
Java 
-------
public class PalindromeChecker{
    public static boolean isPalindrome(String str){
        str = str.toLowerCase();
        int left = 0, right = str.length()-1;
        while (left<right){
            if(str.charAt(left)!=str.charAt(right)){
                return false;
            }
            left++;
            right--;
        }
        return true;
    }
    public static void main(String[] args){
        System.out.println("Palindrome Checker:"+isPalindrome("racecar"));
	System.out.println("Palindrome Checker:"+isPalindrome("hello"));
    } 
}

        int a = 0, b = 1, temp;
        System.out.print("Fibonacci Series:" +a+" "+b+" ");
        for(int i=2; i<=5; i++ ){
            temp = a + b;
            System.out.print(temp+ " ");
            a = b;
            b = temp;
        }
        System.out.println(" ");
    }
}

7. Description: Given an array and a number k, rotate the array to the right by k positions without using predefined methods.

Example:
Input:
arr = [1, 2, 3, 4, 5]
k = 2

Output:
[4, 5, 1, 2, 3]
-----------------------------------------------
Java
-------
package sample_Tasks;

class RotateArray {

	public static void main(String[] args) {
		int[] arr = { 1, 2, 3, 4, 5 };
		int k = 2;
		rotateRight(arr, k);
		for (int num : arr) {
			System.out.print(num + " ");
		}

	}

	public static void rotateRight(int[] arr, int k) {
		int n = arr.length;
		reverse(arr, 0, n - 1);
		reverse(arr, 0, k - 1);
		reverse(arr, k, n - 1);
	}

	public static void reverse(int[] arr, int start, int end) {
		while (start < end) {
			int temp = arr[start];
			arr[start] = arr[end];
			arr[end] = temp;
			start++;
			end--;
		}
	}
}

arr = [2, 4, 3, 5, 6, 7, 8]
target = 9

Output:

package sample_Tasks;

public class ArrayAddUp {

	public static void main(String[] args) {
		int[] arr = {2, 4, 3, 5, 6, 7, 8};
		int target = 9;
		findPairs(arr, target);
	}
	public static void findPairs(int[] arr, int target) {
		int n = arr.length;
		for(int i=0;i<n;i++) {
			for(int j=i+1;j<n;j++) {
				if(arr[i]+arr[j]==target) {
					System.out.println(arr[i]+","+arr[j]);
				}
			}
		}
	}

}
